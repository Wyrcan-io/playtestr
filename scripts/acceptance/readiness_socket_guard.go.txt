package main

import (
	"crypto/sha256"
	"encoding/hex"
	"encoding/json"
	"errors"
	"fmt"
	"io"
	"os"
	"path/filepath"
	"runtime"
	"syscall"
	"unsafe"
)

// This test-only launcher denies socket creation and connection for its exec
// subtree. It changes no firewall or host policy and is not a security sandbox.
func install() error {
	runtime.LockOSThread()
	filter := []syscall.SockFilter{
		{Code: 0x20, K: 4},                 // seccomp_data.arch
		{Code: 0x15, Jt: 1, K: 0xc000003e}, // AUDIT_ARCH_X86_64
		{Code: 0x06, K: 0x80000000},        // reject another syscall ABI
		{Code: 0x20, K: 0},                 // seccomp_data.nr
	}
	for _, number := range []uint32{syscall.SYS_SOCKET, syscall.SYS_SOCKETPAIR, syscall.SYS_CONNECT} {
		filter = append(filter, syscall.SockFilter{Code: 0x15, Jf: 1, K: number},
			syscall.SockFilter{Code: 0x06, K: 0x00050000 | uint32(syscall.EPERM)})
	}
	filter = append(filter, syscall.SockFilter{Code: 0x06, K: 0x7fff0000})
	program := syscall.SockFprog{Len: uint16(len(filter)), Filter: &filter[0]}
	if _, _, errno := syscall.RawSyscall6(syscall.SYS_PRCTL, 38, 1, 0, 0, 0, 0); errno != 0 {
		return fmt.Errorf("set process no-new-privileges: %w", errno)
	}
	if _, _, errno := syscall.RawSyscall(syscall.SYS_PRCTL, 22, 2, uintptr(unsafe.Pointer(&program))); errno != 0 {
		return fmt.Errorf("install process socket filter: %w", errno)
	}
	runtime.KeepAlive(filter)
	return nil
}

func run() error {
	if len(os.Args) == 2 && (os.Args[1] == "--baseline-probe" || os.Args[1] == "--guard-probe") {
		guarded := os.Args[1] == "--guard-probe"
		if guarded {
			if err := install(); err != nil {
				return err
			}
		}
		fd, err := syscall.Socket(syscall.AF_INET, syscall.SOCK_STREAM, 0)
		if guarded {
			if !errors.Is(err, syscall.EPERM) {
				if fd >= 0 {
					_ = syscall.Close(fd)
				}
				return fmt.Errorf("socket probe expected EPERM, got %v", err)
			}
			fmt.Println("socket-denied=EPERM")
			return nil
		}
		if err != nil {
			return fmt.Errorf("baseline socket probe: %w", err)
		}
		if err := syscall.Close(fd); err != nil {
			return err
		}
		fmt.Println("socket-created-and-closed; no connection attempted")
		return nil
	}
	if len(os.Args) < 7 || os.Args[1] != "--root" || os.Args[3] != "--receipt" || os.Args[5] != "--" {
		return errors.New("usage: guard --root OWNED_ROOT --receipt FILE -- EXECUTABLE [ARGS]")
	}
	root, err := filepath.EvalSymlinks(os.Args[2])
	if err != nil {
		return err
	}
	parent, err := filepath.EvalSymlinks(filepath.Dir(os.Args[4]))
	if err != nil {
		return err
	}
	relative, err := filepath.Rel(root, parent)
	if err != nil || relative == ".." || filepath.IsAbs(relative) || (len(relative) > 3 && relative[:3] == "../") {
		return errors.New("receipt escapes owned root")
	}
	if !filepath.IsAbs(os.Args[6]) {
		return errors.New("target executable must be absolute")
	}
	file, err := os.Open(os.Args[6])
	if err != nil {
		return err
	}
	digest := sha256.New()
	n, hashErr := io.Copy(digest, io.LimitReader(file, 32*1024*1024+1))
	closeErr := file.Close()
	if hashErr != nil || closeErr != nil || n > 32*1024*1024 {
		return errors.New("cannot hash bounded target executable")
	}
	if err := install(); err != nil {
		return err
	}
	output, err := os.OpenFile(os.Args[4], os.O_WRONLY|os.O_CREATE|os.O_EXCL, 0600)
	if err != nil {
		return err
	}
	err = json.NewEncoder(output).Encode(map[string]any{
		"schema_version": 1, "filter_applied": true, "architecture": "linux-amd64",
		"pid":             os.Getpid(),
		"denied_syscalls": []string{"socket", "socketpair", "connect"},
		"target_sha256":   hex.EncodeToString(digest.Sum(nil)),
		"scope":           "Process exec subtree only; no firewall changes, OS network isolation or security sandbox claim.",
	})
	if closeErr := output.Close(); err != nil || closeErr != nil {
		return errors.New("write applied-filter receipt")
	}
	return syscall.Exec(os.Args[6], os.Args[6:], os.Environ())
}

func main() {
	if err := run(); err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(125)
	}
}
